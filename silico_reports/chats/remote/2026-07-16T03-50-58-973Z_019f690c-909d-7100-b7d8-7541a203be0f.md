#!/bin/bash
# Driver for one Mizan Rooms SLURM task: start the local Qwen shim, wait for
# health, run the package runner for each requested condition sequentially,
# then stop the shim. Never modifies any file inside the package repo.
#
# Required env:
#   EXP_PROJ   absolute path to the experiment dir (uv project + src/)
#   REPO_DIR   absolute path to the pinned ConstitutionalAlignment clone
#   SPLIT      development | evaluation
#   SEED       generation base seed
#   CONDITIONS space-separated condition ids
#   REPLICATES replicate count
#   OUTPUT_ROOT absolute output root (task dirs created beneath)
#   EXPECTED_REQUESTS  total turns this task will generate (telemetry)
# Optional env:
#   UNSEAL=1 + MIZAN_ANALYSIS_PLAN_SHA256   evaluation gate
#   MODEL_ID (default Qwen/Qwen3.5-9B), MODEL_REVISION
set -euo pipefail

MODEL_ID="${MODEL_ID:-Qwen/Qwen3.5-9B}"
PORT=$((8815 + SEED))
API_BASE="http://127.0.0.1:${PORT}/v1"

echo "driver: split=$SPLIT seed=$SEED conditions=[$CONDITIONS] replicates=$REPLICATES"

uv run --project "$EXP_PROJ" python "$EXP_PROJ/src/qwen_shim.py" \
  --model-id "$MODEL_ID" ${MODEL_REVISION:+--revision "$MODEL_REVISION"} \
  --port "$PORT" --expected-requests "$EXPECTED_REQUESTS" &
SHIM_PID=$!
trap 'kill $SHIM_PID 2>/dev/null || true' EXIT

for i in $(seq 1 180); do
  if curl -sf "http://127.0.0.1:${PORT}/health" >/dev/null 2>&1; then break; fi
  if ! kill -0 $SHIM_PID 2>/dev/null; then echo "shim died during startup" >&2; exit 3; fi
  sleep 5
done
curl -sf "http://127.0.0.1:${PORT}/health" >/dev/null || { echo "shim never became healthy" >&2; exit 3; }
echo "driver: shim healthy on port $PORT"

cd "$REPO_DIR"
for condition in $CONDITIONS; do
  out_dir="$OUTPUT_ROOT/${condition}_s${SEED}"
  args=(
    --condition "$condition"
    --split "$SPLIT"
    --seed "$SEED"
    --replicates "$REPLICATES"
    --blinding-seed 20260716
    --model-id "$MODEL_ID"
    --policy openai-compatible
    --api-base "$API_BASE"
    --temperature 0.2
    --top-p 1.0
    --max-tokens 180
    --timeout-seconds 180
    --output-dir "$out_dir"
  )
  if [[ "${UNSEAL:-0}" == "1" ]]; then
    args+=(--unseal-evaluation --analysis-plan-sha256 "$MIZAN_ANALYSIS_PLAN_SHA256")
  fi
  echo "driver: running condition=$condition -> $out_dir"
  uv run --project "$EXP_PROJ" python scripts/run_mizan_rooms.py "${args[@]}"
done
echo "driver: all conditions complete"
